1//! Writes to and reads from a file or block device at offsets, on the
2//! reactor's ring: the engine's commits (I199) and reads (D190), which used
3//! to wait in the device's own calls and so held the reactor, pings
4//! included, for as long as the device took.
5
6use crate::{Resource, check, current};
7use io_uring::{opcode, types};
8use std::cell::RefCell;
9use std::io;
10use std::ops::DerefMut;
11use std::os::fd::RawFd;
12use std::rc::Rc;
13
14/// Writes every buffer at its byte offset of `fd`, all submitted at once,
15/// then, once each has completed in full, makes them durable
16/// (`fdatasync`). The first error wins; a short write is one.
17///
18/// The buffers are the reactor's until the kernel is done with them, so
19/// dropping this future (a timeout, say) frees nothing the kernel still
20/// uses.
21///
22/// # Safety
23///
24/// `fd` must stay open until every write and the flush have completed,
25/// even if this future is dropped first.
26pub async unsafe fn write_then_sync<B: AsRef<[u8]> + 'static>(
27    fd: RawFd,
28    writes: Vec<(u64, B)>,
29) -> io::Result<()> {
30    let r = current();
31    let bufs = Rc::new(writes);
32    let keep: Rc<dyn std::any::Any> = bufs.clone();
33    let ops: Vec<_> = bufs
34        .iter()
35        .map(|(off, b)| {
36            let b = b.as_ref();
37            let sqe = opcode::Write::new(types::Fd(fd), b.as_ptr(), b.len() as u32)
38                .offset(*off)
39                .build();
40            (r.submit(sqe, Resource::Keep(keep.clone())), b.len())
41        })
42        .collect();
43    drop((bufs, keep));
44    let mut err = None;
45    for (op, want) in ops {
46        let (n, _) = op.await;
47        match check(n) {
48            Ok(n) if n as usize == want => {}
49            Ok(n) => {
50                err.get_or_insert(io::Error::new(
51                    io::ErrorKind::UnexpectedEof,
52                    format!("short write: {n} of {want} bytes"),
53                ));
54            }
55            Err(e) => {
56                err.get_or_insert(e);
57            }
58        }
59    }
60    if let Some(e) = err {
61        return Err(e);
62    }
63    let sqe = opcode::Fsync::new(types::Fd(fd))
64        .flags(types::FsyncFlags::DATASYNC)
65        .build();
66    let (n, _) = r.submit(sqe, Resource::None).await;
67    check(n).map(|_| ())
68}

70/// Reads into `buf` at each `(byte offset of fd, offset in buf, length)`,
71/// all submitted at once, and gives `buf` back once each read has
72/// completed in full. The first error wins; a short read is one.
73///
74/// As for [`write_then_sync`], the buffer is the reactor's until the kernel
75/// is done with it, so dropping this future frees nothing the kernel still
76/// writes to.
77///
78/// # Safety
79///
80/// `fd` must stay open until every read has completed, even if this future
81/// is dropped first.
82pub async unsafe fn read_into<B: DerefMut<Target = [u8]> + 'static>(
83    fd: RawFd,
84    buf: B,
85    ios: &[(u64, usize, usize)],
86) -> io::Result<B> {
87    let r = current();
88    let cell = Rc::new(RefCell::new(buf));
89    let base = {
90        let mut b = cell.borrow_mut();
91        for &(_, at, len) in ios {
92            assert!(at + len <= b.len(), "a read past the buffer");
93        }
94        b.as_mut_ptr()
95    };
96    let keep: Rc<dyn std::any::Any> = cell.clone();
97    let ops: Vec<_> = ios
98        .iter()
99        .map(|&(off, at, len)| {
100            // SAFETY: `at + len` is within the buffer (asserted above), whose
101            // memory stays where it is inside `cell`, which the ring keeps
102            // until this read completes; nothing else touches the buffer
103            // until every read has completed.
104            let ptr = unsafe { base.add(at) };
105            let sqe = opcode::Read::new(types::Fd(fd), ptr, len as u32)
106                .offset(off)
107                .build();
108            (r.submit(sqe, Resource::Keep(keep.clone())), len)
109        })
110        .collect();
111    drop(keep);
112    let mut err = None;
113    for (op, want) in ops {
114        let (n, res) = op.await;
115        drop(res);
116        match check(n) {
117            Ok(n) if n as usize == want => {}
118            Ok(n) => {
119                err.get_or_insert(io::Error::new(
120                    io::ErrorKind::UnexpectedEof,
121                    format!("short read: {n} of {want} bytes"),
122                ));
123            }
124            Err(e) => {
125                err.get_or_insert(e);
126            }
127        }
128    }
129    if let Some(e) = err {
130        return Err(e);
131    }
132    let Ok(cell) = Rc::try_unwrap(cell) else {
133        unreachable!("every read has completed and given its hold back");
134    };
135    Ok(cell.into_inner())
136}

138#[cfg(test)]
139mod tests {
140    use super::*;
141    use std::os::fd::AsRawFd;
142    use std::os::unix::fs::FileExt;

144    #[test]
145    fn writes_land_at_their_offsets() {
146        let dir = std::env::temp_dir().join(format!("hs-file-{}", std::process::id()));
147        std::fs::create_dir_all(&dir).unwrap();
148        let path = dir.join("dev");
149        let f = std::fs::OpenOptions::new()
150            .read(true)
151            .write(true)
152            .create(true)
153            .truncate(true)
154            .open(&path)
155            .unwrap();
156        f.set_len(1 << 16).unwrap();
157        let fd = f.as_raw_fd();
158        crate::block_on(async move {
159            let w = vec![(4096u64, vec![1u8; 4096]), (12288, vec![2u8; 8192])];
160            // SAFETY: `f` stays open until after block_on returns.
161            unsafe { write_then_sync(fd, w) }.await.unwrap();
162        });
163        let mut b = vec![0u8; 1 << 16];
164        f.read_exact_at(&mut b, 0).unwrap();
165        assert!(b[..4096].iter().all(|&x| x == 0));
166        assert!(b[4096..8192].iter().all(|&x| x == 1));
167        assert!(b[8192..12288].iter().all(|&x| x == 0));
168        assert!(b[12288..20480].iter().all(|&x| x == 2));
169        drop(f);
170        std::fs::remove_dir_all(&dir).unwrap();
171    }
172
173    #[test]
174    fn reads_land_in_the_buffer() {
175        let dir = std::env::temp_dir().join(format!("hs-read-{}", std::process::id()));
176        std::fs::create_dir_all(&dir).unwrap();
177        let path = dir.join("dev");
178        let f = std::fs::File::create_new(&path).unwrap();
179        let mut b = vec![0u8; 1 << 16];
180        b[4096..8192].fill(1);
181        b[20480..28672].fill(2);
182        f.write_all_at(&b, 0).unwrap();
183        let fd = f.as_raw_fd();
184        let (got, short) = crate::block_on(async move {
185            // SAFETY: `f` stays open until after block_on returns.
186            let got = unsafe {
187                read_into(
188                    fd,
189                    vec![9u8; 16384],
190                    &[(4096, 0, 4096), (20480, 8192, 8192)],
191                )
192            }
193            .await
194            .unwrap();
195            // SAFETY: as above.
196            let short = unsafe { read_into(fd, vec![0u8; 8192], &[(61440, 0, 8192)]) }.await;
197            (got, short)
198        });
199        assert!(got[..4096].iter().all(|&x| x == 1));
200        assert!(got[4096..8192].iter().all(|&x| x == 9), "left alone");
201        assert!(got[8192..].iter().all(|&x| x == 2));
202        assert_eq!(short.unwrap_err().kind(), io::ErrorKind::UnexpectedEof);
203        drop(f);
204        std::fs::remove_dir_all(&dir).unwrap();
205    }

207    #[test]
208    fn errors_are_returned() {
209        crate::block_on(async {
210            // SAFETY: -1 is no descriptor; the kernel refuses it at once.
211            let r = unsafe { write_then_sync(-1, vec![(0u64, vec![0u8; 4096])]) }.await;
212            assert_eq!(r.unwrap_err().raw_os_error(), Some(libc::EBADF));
213            // SAFETY: as above.
214            let r = unsafe { read_into(-1, vec![0u8; 4096], &[(0, 0, 4096)]) }.await;
215            assert_eq!(r.unwrap_err().raw_os_error(), Some(libc::EBADF));
216        });
217    }
