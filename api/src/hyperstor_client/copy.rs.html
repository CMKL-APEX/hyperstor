1//! Copies and moves between pools (10-volumes.md §10.6, 04-decisions.md
2//! D214).
3//!
4//! A copy reads a volume's head (the volume held meanwhile, so nothing
5//! writes it) or one of its snapshots, and writes what it reads into a new
6//! volume in another pool, held too, skipping holes and blocks of zeros. The
7//! copy is flat: no snapshots, no parents. Pools may differ in chunk size:
8//! data is read and written by offset.
9//!
10//! A move keeps the volume's id. Its head is copied into lineage `id` of
11//! the new pool (free there: a lineage is named per pool), then its header
12//! points there in one compare-and-set, and the data left in the old pool
13//! is deleted. Before the switch, what is left behind is recorded in
14//! `m/{id}` in `.meta`, so a move that stops after it is finished by
15//! running it again. A move needs the volume detached, without snapshots,
16//! and with nothing reading through it; it stays held until the old data
17//! is gone.
18

29/// Bytes read and written at a time, and how many such steps are in flight.
30const STEP: u64 = 4 << 20;
31const IN_FLIGHT: u64 = 8;
32
33const OLD: &[u8] = b"old";
34
35/// How far a copy or move has got, and whether to stop: shared with
36/// whoever follows it (the map service's operation, on another thread).
37#[derive(Debug, Default)]
38pub struct Progress {
39    /// Bytes of the volume gone through.
40    pub done: AtomicU64,
41    /// The volume's size.
42    pub total: AtomicU64,
43    pub cancel: AtomicBool,
44    /// For tests: a move stops right after its switch, as if the map
45    /// service had stopped there.
46    #[doc(hidden)]
47    pub stop_after_switch: AtomicBool,
48}
49
50/// Where copied blocks go.
51#[derive(Clone)]
52enum Dest {
53    /// A new volume, held by the copy.
54    Volume(Rc<Attached>),
55    /// Lineage `lineage` of `pool`, at `epoch`, under lease `lease`.
56    Raw {
57        lease: u64,
58        pool: u32,
59        lineage: u128,
60        epoch: u64,
61    },
62}
63
64/// The data a move leaves in the old pool.
65#[derive(Debug, Clone, PartialEq, Eq)]
66struct Left {
67    to: u32,
68    pool: u32,
69    lineage: u128,
70    size: u64,
71    branches: Vec<u32>,
72}
73
74fn moving_key(id: u128) -> Vec<u8> {
75    format!("m/{id:032x}").into_bytes()
76}
77
78fn encode_left(l: &Left) -> Vec<u8> {
79    let mut e = Enc::new();
80    e.u32(l.to).u32(l.pool).u128(l.lineage).u64(l.size);
81    e.seq(&l.branches, |e, b| {
82        e.u32(*b);
83    });
84    e.into_vec()
85}
86
87fn decode_left(b: &[u8]) -> Option<Left> {
88    let mut d = Dec::new(b);
89    Some(Left {
90        to: d.u32().ok()?,
91        pool: d.u32().ok()?,
92        lineage: d.u128().ok()?,
93        size: d.u64().ok()?,
94        branches: d.seq(4, |d| d.u32()).ok()?,
95    })
96}
97
98fn cancelled(p: &Progress) -> Result<(), ClientError> {
99    if p.cancel.load(Ordering::Relaxed) {
100        return Err(ClientError::Refused("cancelled".into()));
101    }
102    Ok(())
103}

105impl Client {
106    /// Takes volume `id`'s lease for a copy or a move: from a holder
107    /// that has stopped renewing too, such as a former leader that was
108    /// running the same operation (it is blocklisted, D125); a holder
109    /// that renews keeps it.
110    async fn hold(self: &Rc<Self>, id: u128) -> Result<Rc<Attached>, ClientError> {
111        self.attach_as(id, Take::TakeOver, &self.lease_label())
112            .await
113    }
114
115    /// Copies volume `id` into a new volume `new_id` named `name` in pool
116    /// `pool`: its snapshot at `epoch`, or its head, which needs it
117    /// detached (it is held meanwhile). The new volume is held while it
118    /// fills, and deleted if the copy fails. Running it again after it
119    /// stopped starts over, into the same new volume.
120    pub async fn copy_volume(
121        self: &Rc<Self>,
122        id: u128,
123        epoch: Option<u64>,
124        pool: u32,
125        new_id: u128,
126        name: &str,
127        p: &Progress,
128    ) -> Result<Volume, ClientError> {
129        let v = self.volume(id).await?.ok_or(ClientError::NoVolume(id))?;
130        if v.state != VolumeState::Active {
131            return Err(ClientError::NoVolume(id));
132        }
133        let held = match epoch {
134            Some(_) => None,
135            None => Some(self.hold(id).await.map_err(|e| attached(id, e))?),
136        };
137        let r = self.copy_held(id, epoch, pool, new_id, name, p).await;
138        if let Some(h) = held {
139            let _ = h.detach().await;
140        }
141        r
142    }
143
144    async fn copy_held(
145        self: &Rc<Self>,
146        id: u128,
147        epoch: Option<u64>,
148        pool: u32,
149        new_id: u128,
150        name: &str,
151        p: &Progress,
152    ) -> Result<Volume, ClientError> {
153        let v = self.volume(id).await?.ok_or(ClientError::NoVolume(id))?;
154        let layers = match epoch {
155            Some(e) => {
156                let s = v
157                    .snapshot(e)
158                    .filter(|s| !s.deleting && s.trashed_unix == 0)
159                    .ok_or(ClientError::NoSnapshot(e))?;
160                v.snapshot_layers(s)
161            }
162            None => v.layers(),
163        };
164        // A copy run again finds its volume: emptied, then filled again.
165        match self.volume(new_id).await? {
166            Some(n) if n.pool == pool && n.size == v.size && n.lineage == new_id => {
167                self.drop_data(pool, new_id, n.size, &[0]).await?;
168            }
169            Some(_) => return Err(ClientError::VolumeExists(new_id)),
170            None => {
171                self.create_volume(pool, v.size, Some(new_id), name).await?;
172            }
173        }
174        let r = async {
175            let to = self.hold(new_id).await?;
176            let r = self
177                .copy_blocks(&v, &layers, &Dest::Volume(to.clone()), p)
178                .await;
179            let _ = to.detach().await;
180            r?;
181            // A snapshot deleted meanwhile may have been merged under the
182            // copy: refused, as if it had gone before.
183            if let Some(e) = epoch {
184                let now = self.volume(id).await?.ok_or(ClientError::NoVolume(id))?;
185                now.snapshot(e)
186                    .filter(|s| !s.deleting && s.trashed_unix == 0)
187                    .ok_or(ClientError::NoSnapshot(e))?;
188            }
189            self.volume(new_id)
190                .await?
191                .ok_or(ClientError::NoVolume(new_id))
192        }
193        .await;
194        if r.is_err() {
195            let _ = self.delete_volume(new_id).await;
196        }
197        r
198    }

200    /// Moves volume `id` to pool `pool`, keeping its id. Refused while it
201    /// is attached, has snapshots (in the trash too), or another volume
202    /// reads through it; a clone is flattened by the move. Running it
203    /// again after it stopped finishes it, or starts over if the switch
204    /// wasn't made.
205    pub async fn move_volume(
206        self: &Rc<Self>,
207        id: u128,
208        pool: u32,
209        p: &Progress,
210    ) -> Result<Volume, ClientError> {
211        let mut v = self.volume(id).await?.ok_or(ClientError::NoVolume(id))?;
212        if v.state != VolumeState::Active {
213            return Err(ClientError::NoVolume(id));
214        }
215        if !self.map().pools.contains_key(&pool) || pool == META_POOL {
216            return Err(ClientError::NoPool(pool));
217        }
218        let left = self.left(id).await?;
219        if v.pool == pool && left.is_none() {
220            return Ok(v);
221        }
222        let held = self.hold(id).await.map_err(|e| attached(id, e))?;
223        let r = async {
224            match left {
225                // Switched before it stopped: what is left goes.
226                Some(l) if v.pool == l.to => {}
227                // Not switched: as if it hadn't started.
228                Some(_) => {
229                    self.exec(META_POOL, NS_META, &moving_key(id), &[WireOp::Delete])
230                        .await?;
231                    v = self.switch_to(&held, pool, p).await?;
232                }
233                None => v = self.switch_to(&held, pool, p).await?,
234            }
235            if p.stop_after_switch.load(Ordering::Relaxed) {
236                return Err(ClientError::Refused("stopped after the switch".into()));
237            }
238            self.finish_move(id).await?;
239            self.volume(id).await?.ok_or(ClientError::NoVolume(id))
240        }
241        .await;
242        let _ = held.detach().await;
243        r
244    }

246    /// Copies held volume `held` into lineage `id` of `pool`, records what
247    /// is left behind and switches the header over.
248    async fn switch_to(
249        self: &Rc<Self>,
250        held: &Rc<Attached>,
251        pool: u32,
252        p: &Progress,
253    ) -> Result<Volume, ClientError> {
254        let v = self.volume(held.volume().id).await?;
255        let v = v.ok_or(ClientError::NoVolume(held.volume().id))?;
256        let id = v.id;
257        if !v.snapshots.is_empty() {
258            return Err(ClientError::Refused(format!(
259                "volume {id:032x} has snapshots: delete and purge them first"
260            )));
261        }
262        let (own, branches) = self.own_branches(&v).await?;
263        for b in branches.iter().filter(|b| b.volume != id) {
264            if let Some((_, e)) = b.chain.iter().find(|(x, _)| own.contains(x))
265                && self.branch_reads(b).await?
266            {
267                return Err(ClientError::SnapshotInUse(*e));
268            }
269        }
270        // What an earlier attempt may have left there goes first: blocks
271        // of zeros aren't written.
272        self.drop_data(pool, id, v.size, &[0]).await?;
273        let dest = Dest::Raw {
274            lease: held.lease(),
275            pool,
276            lineage: id,
277            epoch: v.head_epoch,
278        };
279        let r = self.copy_blocks(&v, &v.layers(), &dest, p).await;
280        if let Err(e) = r {
281            let _ = self.drop_data(pool, id, v.size, &[0]).await;
282            return Err(e);
283        }
284        let left = Left {
285            to: pool,
286            pool: v.pool,
287            lineage: v.lineage,
288            size: v.size,
289            branches: own,
290        };
291        self.exec(
292            META_POOL,
293            NS_META,
294            &moving_key(id),
295            &[
296                WireOp::Touch,
297                WireOp::SetAttr(OLD.to_vec(), encode_left(&left)),
298            ],
299        )
300        .await?;
301        let new = Volume {
302            pool,
303            lineage: id,
304            branch: 0,
305            chain: Vec::new(),
306            skip: Vec::new(),
307            incarnation: random_u64() | 1,
308            version: v.version + 1,
309            ..v.clone()
310        };
311        // Held by this lease, unchanged since it was read: the copy is
312        // what it holds.
313        let mut ops = cas(v.version, &new);
314        ops.insert(
315            0,
316            WireOp::AssertAttr {
317                name: crate::volume::LEASE_ATTR.to_vec(),
318                value: Some(held.holder_attr()),
319            },
320        );
321        match self.exec(META_POOL, NS_META, &header_key(id), &ops).await {
322            Ok(()) => Ok(new),
323            Err(e) => {
324                let _ = self
325                    .exec(META_POOL, NS_META, &moving_key(id), &[WireOp::Delete])
326                    .await;
327                let _ = self.drop_data(pool, id, v.size, &[0]).await;
328                if conflict(&e) {
329                    return Err(ClientError::Refused(format!(
330                        "volume {id:032x} changed during the move"
331                    )));
332                }
333                Err(e)
334            }
335        }
336    }

338    /// After the switch: lists the volume in its new pool's directory,
339    /// takes it from the old one's, deletes the data left there and the
340    /// branch records it owned, then the record of the move.
341    async fn finish_move(self: &Rc<Self>, id: u128) -> Result<(), ClientError> {
342        let Some(l) = self.left(id).await? else {
343            return Ok(());
344        };
345        self.exec(
346            META_POOL,
347            NS_META,
348            &dir_key(l.to, id),
349            &[
350                WireOp::Touch,
351                WireOp::KvPut(id.to_be_bytes().to_vec(), Vec::new()),
352            ],
353        )
354        .await?;
355        self.exec(
356            META_POOL,
357            NS_META,
358            &dir_key(l.pool, id),
359            &[WireOp::Touch, WireOp::KvDel(id.to_be_bytes().to_vec())],
360        )
361        .await?;
362        self.drop_data(l.pool, l.lineage, l.size, &l.branches)
363            .await?;
364        for b in l.branches.iter().filter(|b| **b != 0) {
365            self.forget_branch(l.lineage, *b).await?;
366        }
367        self.exec(META_POOL, NS_META, &moving_key(id), &[WireOp::Delete])
368            .await
369    }

371    /// What a move of `id` recorded it leaves behind, if one is under way.
372    async fn left(&self, id: u128) -> Result<Option<Left>, ClientError> {
373        let l = self
374            .lookup(META_POOL, NS_META, &moving_key(id), b"", None, 0)
375            .await?;
376        Ok(l.and_then(|l| attr(&l, OLD).and_then(decode_left)))
377    }

379    /// Deletes every version of `branches` of lineage `lineage` in `pool`,
380    /// over `size` bytes, a few chunks at a time.
381    async fn drop_data(
382        self: &Rc<Self>,
383        pool: u32,
384        lineage: u128,
385        size: u64,
386        branches: &[u32],
387    ) -> Result<(), ClientError> {
388        let chunk = u64::from(
389            self.map()
390                .pools
391                .get(&pool)
392                .ok_or(ClientError::NoPool(pool))?
393                .chunk_size,
394        );
395        let chunks = size.div_ceil(chunk);
396        for b in branches {
397            let mut next = 0;
398            while next < chunks {
399                let batch: Vec<_> = (next..(next + 32).min(chunks))
400                    .map(|c| {
401                        let (me, b) = (self.clone(), *b);
402                        hyperstor_reactor::spawn(async move {
403                            me.drop_versions(pool, lineage, c, b, 0, u64::MAX).await
404                        })
405                    })
406                    .collect();
407                for x in batch {
408                    x.await?;
409                }
410                next += 32;
411            }
412        }
413        Ok(())
414    }

416    /// Reads `v` through `layers` and writes what isn't a hole or zeros
417    /// to `dest`, [`STEP`] bytes at a time, [`IN_FLIGHT`] steps at once.
418    async fn copy_blocks(
419        self: &Rc<Self>,
420        v: &Volume,
421        layers: &[(u32, u64, u64)],
422        dest: &Dest,
423        p: &Progress,
424    ) -> Result<(), ClientError> {
425        p.total.store(v.size, Ordering::Relaxed);
426        p.done.store(0, Ordering::Relaxed);
427        let layers: Rc<Vec<(u32, u64, u64)>> = Rc::new(layers.to_vec());
428        let dest = Rc::new(dest.clone());
429        let mut off = 0;
430        while off < v.size {
431            cancelled(p)?;
432            let batch: Vec<_> = (0..IN_FLIGHT)
433                .map(|i| off + i * STEP)
434                .take_while(|o| *o < v.size)
435                .map(|o| {
436                    let len = STEP.min(v.size - o);
437                    let (me, layers, dest) = (self.clone(), layers.clone(), dest.clone());
438                    let (pool, lineage) = (v.pool, v.lineage);
439                    hyperstor_reactor::spawn(async move {
440                        let (data, sources) =
441                            me.read_sourced(pool, lineage, &layers, o, len).await?;
442                        me.write_runs(&dest, o, &data, &sources).await
443                    })
444                })
445                .collect();
446            let n = batch.len() as u64;
447            for b in batch {
448                b.await?;
449            }
450            off = (off + n * STEP).min(v.size);
451            p.done.store(off, Ordering::Relaxed);
452        }
453        Ok(())
454    }

456    /// Writes each run of blocks of `data` (read at `off`) that holds
457    /// something: not a hole, not zeros.
458    async fn write_runs(
459        self: &Rc<Self>,
460        dest: &Dest,
461        off: u64,
462        data: &[u8],
463        sources: &[(u32, u64)],
464    ) -> Result<(), ClientError> {
465        let blocks = data.len() / BLOCK as usize;
466        let full = |i: usize| {
467            sources.get(i).is_none_or(|s| s.0 != u32::MAX)
468                && data[i * BLOCK as usize..(i + 1) * BLOCK as usize]
469                    .iter()
470                    .any(|b| *b != 0)
471        };
472        let mut i = 0;
473        while i < blocks {
474            if !full(i) {
475                i += 1;
476                continue;
477            }
478            let start = i;
479            while i < blocks && full(i) {
480                i += 1;
481            }
482            let at = off + (start as u64) * BLOCK;
483            let run = data[start * BLOCK as usize..i * BLOCK as usize].to_vec();
484            match dest {
485                Dest::Volume(a) => a.write(at, run).await?,
486                Dest::Raw {
487                    lease,
488                    pool,
489                    lineage,
490                    epoch,
491                } => {
492                    self.write_under(*lease, *pool, *lineage, 0, *epoch, at, run)
493                        .await?
494                }
495            }
496        }
497        Ok(())
498    }
499}

501/// A volume someone holds, as the API names it.
502fn attached(id: u128, e: ClientError) -> ClientError {
503    match e {
504        ClientError::LeaseHeld { .. } => ClientError::VolumeAttached(id),
505        e => e,
506    }
