1//! Snapshots, clones, rollback and flatten (10-volumes.md §10.3, §10.4;
2//! 04-decisions.md D10, D21, D101, D103, D104).
3//!
4//! A chunk's data in a branch is a set of versions, one per head epoch it
5//! was written in (08-engine.md §7.2). A snapshot is a compare-and-set on the
6//! header: it records the head epoch, and the head moves on; nothing is
7//! copied. A clone or a rollback is a new branch of the lineage, cut from
8//! a snapshot; its reads go through its chain of parents.
9//!
10//! The lineage object `l/{lineage}` gives out branch numbers and records
11//! each branch (as key–value entries): its volume, the chain it was cut
12//! with, and the epoch it was flattened at, if it was. A snapshot is in
13//! use while a branch that still reads its chain has it in its chain; a
14//! flattened branch still does while its volume has snapshots from before
15//! the flatten (01-implementation-notes.md I115). Clone and delete race
16//! safely: a delete marks the snapshot deleting before it looks for
17//! dependents, and a clone records itself before it checks the snapshot
18//! is still live.
19

27/// At most this many snapshots per volume (10-volumes.md §10.3).
28pub const MAX_SNAPSHOTS: usize = 255;
29/// Clone chains longer than this are refused: flatten first.
30pub const MAX_CHAIN: usize = 16;

32const NEXT: &[u8] = b"next";
33const STATE: &[u8] = b"state";
34const MEMBERS: &[u8] = b"members";

36fn group_key(gid: u128) -> Vec<u8> {
37    format!("g/{gid:032x}").into_bytes()
38}

40fn lineage_key(lineage: u128) -> Vec<u8> {
41    format!("l/{lineage:032x}").into_bytes()
42}

44/// A branch of a lineage, as its lineage object records it.
45#[derive(Debug, Clone, PartialEq, Eq)]
46pub struct Branch {
47    pub branch: u32,
48    /// The volume whose branch it is.
49    pub volume: u128,
50    /// The chain it was cut with: its parent first. Kept as it was.
51    pub chain: Vec<(u32, u64)>,
52    /// The head epoch it was flattened at (0: not flattened). Reads at that
53    /// epoch or later don't go through the chain.
54    pub flattened: u64,
55}
56
57fn encode_branch(b: &Branch) -> Vec<u8> {
58    let mut e = Enc::new();
59    e.u128(b.volume)
60        .u64(b.flattened)
61        .seq(&b.chain, |e, (x, ep)| {
62            e.u32(*x).u64(*ep);
63        });
64    e.into_vec()
65}
66
67fn decode_branch(k: &[u8], v: &[u8]) -> Option<Branch> {
68    let branch = u32::from_be_bytes(k.try_into().ok()?);
69    let mut d = Dec::new(v);
70    Some(Branch {
71        branch,
72        volume: d.u128().ok()?,
73        flattened: d.u64().ok()?,
74        chain: d.seq(12, |d| Ok((d.u32()?, d.u64()?))).ok()?,
75    })
76}

78fn unexpected(m: Msg) -> ClientError {
79    ClientError::Rpc(crate::RpcError::Protocol(format!("unexpected {m:?}")))
80}

83    /// Whether branch record `r` still reads through its chain: it isn't
84    /// flattened, or its volume has snapshots of it from before.
85    async fn needs_chain(&self, r: &Branch) -> Result<bool, ClientError> {
86        if r.flattened == 0 {
87            return Ok(true);
88        }
89        Ok(match self.volume(r.volume).await? {
90            Some(v) => v
91                .snapshots
92                .iter()
93                .any(|s| s.branch == r.branch && s.epoch < r.flattened),
94            None => false,
95        })
96    }

99    pub(crate) async fn branch_reads(&self, r: &Branch) -> Result<bool, ClientError> {
100        self.needs_chain(r).await
101    }

103    /// The branch records that still read through their chains.
104    async fn reading(&self, branches: &[Branch]) -> Result<Vec<Branch>, ClientError> {
105        let mut out = Vec::new();
106        for b in branches {
107            if self.needs_chain(b).await? {
108                out.push(b.clone());
109            }
110        }
111        Ok(out)
112    }

114    /// The epochs of `branch` something reads at, other than `but`: the
115    /// owner's snapshots of it, its head if on it, and every chain that
116    /// goes through it.
117    fn cuts(v: &Volume, reading: &[Branch], branch: u32, but: u64) -> Vec<u64> {
118        let mut cuts: Vec<u64> = v
119            .snapshots
120            .iter()
121            .filter(|x| x.branch == branch && x.epoch != but && !x.deleting)
122            .map(|x| x.epoch)
123            .chain(
124                reading
125                    .iter()
126                    .flat_map(|r| r.chain.iter())
127                    .filter(|(b, e)| *b == branch && *e != but)
128                    .map(|(_, e)| *e),
129            )
130            .chain(
131                v.snapshots
132                    .iter()
133                    .filter(|x| !x.deleting && x.epoch != but)
134                    .flat_map(|x| x.chain.iter())
135                    .filter(|(b, _)| *b == branch)
136                    .map(|(_, e)| *e),
137            )
138            .collect();
139        if v.branch == branch {
140            cuts.push(v.head_epoch);
141        }
142        cuts.extend(
143            v.chain
144                .iter()
145                .filter(|(b, _)| *b == branch)
146                .map(|(_, e)| *e),
147        );
148        cuts
149    }

151    /// The branches of a lineage other than branch 0, as recorded.
152    pub async fn branches(&self, lineage: u128) -> Result<Vec<Branch>, ClientError> {
153        let l = self
154            .lookup(
155                META_POOL,
156                NS_META,
157                &lineage_key(lineage),

159                None,
160                1 << 20,

171    /// A new branch number in `lineage`.
172    async fn new_branch(&self, lineage: u128) -> Result<u32, ClientError> {
173        let key = lineage_key(lineage);
174        for _ in 0..20 {
175            let cur = self
176                .lookup(META_POOL, NS_META, &key, b"", None, 0)
177                .await?
178                .and_then(|l| attr(&l, NEXT).map(<[u8]>::to_vec));
179            let next: u32 = cur
180                .as_ref()
181                .and_then(|v| std::str::from_utf8(v).ok()?.parse().ok())
182                .unwrap_or(1);
183            let ops = [
184                WireOp::AssertAttr {
185                    name: NEXT.to_vec(),
186                    value: cur,
187                },
188                WireOp::Touch,
189                WireOp::SetAttr(NEXT.to_vec(), (next + 1).to_string().into_bytes()),
190            ];
191            match self.exec(META_POOL, NS_META, &key, &ops).await {
192                Ok(()) => return Ok(next),
193                Err(e) if conflict(&e) => continue,
194                Err(e) => return Err(e),
195            }
196        }
197        Err(ClientError::Refused(format!(
198            "lineage {lineage:032x}: branch numbers kept changing"
199        )))
200    }

229    /// Changes a header by compare-and-set, retrying on conflicts: `f`
230    /// gets the volume as it is and returns the changed one, or an error.
231    async fn change(
232        &self,
233        id: u128,
234        mut f: impl FnMut(&Volume) -> Result<Volume, ClientError>,
235        extra: &[WireOp],
236    ) -> Result<Volume, ClientError> {
237        for _ in 0..10 {
238            let v = self.volume(id).await?.ok_or(ClientError::NoVolume(id))?;
239            let mut new = f(&v)?;
240            new.version = v.version + 1;
241            let mut ops = extra.to_vec();
242            ops.extend(cas(v.version, &new));
243            match self.exec(META_POOL, NS_META, &header_key(id), &ops).await {
244                Ok(()) => return Ok(new),
245                Err(e) if conflict(&e) => continue,
246                Err(e) => return Err(e),
247            }
248        }
249        Err(ClientError::BadVolume(format!(
250            "volume {id:032x} kept changing"
251        )))
252    }

254    /// Takes a snapshot of volume `id` (10-volumes.md §10.3): its head epoch is
255    /// recorded and moves on. If the volume is attached, its holder is
256    /// quiesced first by the header's primary (D103); if it doesn't within
257    /// 2 s the snapshot is abandoned (`Busy`).
258    pub async fn snapshot(self: &Rc<Self>, id: u128, name: &str) -> Result<Snapshot, ClientError> {
259        self.finish_trims(id).await?;
260        self.take_snapshot(id, name, 0, 0).await
261    }

263    /// A snapshot's compare-and-set through the header's primary: on its
264    /// own (`token` 0: the primary quiesces the holder), or under a
265    /// group's hold, tagged with `group`.
266    async fn take_snapshot(
267        &self,
268        id: u128,
269        name: &str,
270        group: u128,
271        token: u64,
272    ) -> Result<Snapshot, ClientError> {
273        let key = header_key(id);
274        for _ in 0..10 {
275            let v = self.volume(id).await?.ok_or(ClientError::NoVolume(id))?;
276            if v.state != VolumeState::Active {
277                return Err(ClientError::NoVolume(id));
278            }
279            if v.snapshots.len() >= MAX_SNAPSHOTS {
280                return Err(ClientError::Refused(format!(
281                    "volume {id:032x} has {MAX_SNAPSHOTS} snapshots"
282                )));
283            }
284            let snap = Snapshot {
285                branch: v.branch,
286                epoch: v.head_epoch,
287                name: name.to_string(),
288                group,
289                deleting: false,
290                chain: v.chain.clone(),
291                trashed_unix: 0,
292            };
293            let mut new = v.clone();
294            new.snapshots.push(snap.clone());
295            new.head_epoch += 1;
296            new.version += 1;
297            let payload = encode_ops(&cas(v.version, &new));
298            let route = Route::Object {
299                ns: NS_META,
300                name: &key,
301            };
302            match self
303                .to_primary(META_POOL, route, 0, &payload, &|to, key| Msg::Snapshot {
304                    to,
305                    key,
306                    token,
307                })
308                .await
309            {
310                Ok((Msg::ExecOk, _)) => return Ok(snap),
311                Ok((m, _)) => return Err(unexpected(m)),
312                Err(e) if conflict(&e) => continue,
313                Err(e) => return Err(e),
314            }
315        }
316        Err(ClientError::BadVolume(format!(
317            "volume {id:032x} kept changing"
318        )))
319    }

321    /// Takes a snapshot of every volume in `ids` at one moment (a
322    /// consistency group, 10-volumes.md §10.3): the group is recorded as
323    /// preparing; every member's holder is quiesced and held, so no member
324    /// writes after any is cut; each member's snapshot is taken, tagged
325    /// with the group; the group is marked complete. If any step fails, the
326    /// snapshots taken are deleted, the holds released and the group
327    /// removed. Returns the group's id and the snapshots.
328    pub async fn snapshot_group(
329        self: &Rc<Self>,
330        ids: &[u128],
331        name: &str,
332    ) -> Result<(u128, Vec<(u128, Snapshot)>), ClientError> {
333        let gid = (u128::from(random_u64()) << 64) | u128::from(random_u64());
334        let token = random_u64().max(1);
335        for id in ids {
336            self.finish_trims(*id).await?;
337        }
338        let members: Vec<u8> = ids.iter().flat_map(|i| i.to_be_bytes()).collect();
339        self.exec(
340            META_POOL,
341            NS_META,
342            &group_key(gid),
343            &[
344                WireOp::AssertAbsent,
345                WireOp::SetAttr(STATE.to_vec(), b"preparing".to_vec()),
346                WireOp::SetAttr(MEMBERS.to_vec(), members),
347                WireOp::SetAttr(b"name".to_vec(), name.as_bytes().to_vec()),
348            ],
349        )
350        .await?;
351        let header = |id: u128| header_key(id);
352        let call = |id: u128, make: &'static dyn Fn(Target, WireKey, u64) -> Msg| {
353            let me = self.clone();
354            async move {
355                let key = header(id);
356                let route = Route::Object {
357                    ns: NS_META,
358                    name: &key,
359                };
360                match me
361                    .to_primary(META_POOL, route, 0, &[], &|to, key| make(to, key, token))
362                    .await?
363                {
364                    (Msg::ExecOk, _) => Ok(()),
365                    (m, _) => Err(unexpected(m)),
366                }
367            }
368        };
369        let quiesce: &'static dyn Fn(Target, WireKey, u64) -> Msg =
370            &|to, key, token| Msg::Quiesce { to, key, token };
371        let release: &'static dyn Fn(Target, WireKey, u64) -> Msg =
372            &|to, key, token| Msg::Release { to, key, token };
373        // Every holder held before any member is cut.
374        let held: Vec<_> = ids
375            .iter()
376            .map(|id| hyperstor_reactor::spawn(call(*id, quiesce)))
377            .collect();
378        let mut failed = None;
379        for h in held {
380            if let Err(e) = h.await {
381                failed.get_or_insert(e);
382            }
383        }
384        let mut taken = Vec::new();
385        if failed.is_none() {
386            for id in ids {
387                match self.take_snapshot(*id, name, gid, token).await {
388                    Ok(s) => taken.push((*id, s)),
389                    Err(e) => {
390                        failed = Some(e);
391                        break;
392                    }
393                }
394            }
395        }
396        if let Some(e) = failed {
397            for id in ids {
398                let _ = call(*id, release).await;
399            }
400            for (id, s) in &taken {
401                let _ = self.delete_snapshot(*id, s.epoch).await;
402            }
403            let _ = self
404                .exec(META_POOL, NS_META, &group_key(gid), &[WireOp::Delete])
405                .await;
406            return Err(e);
407        }
408        self.exec(
409            META_POOL,
410            NS_META,
411            &group_key(gid),
412            &[WireOp::SetAttr(STATE.to_vec(), b"complete".to_vec())],
413        )
414        .await?;
415        Ok((gid, taken))
416    }

418    /// Whether group `gid` is complete; `None` if there is no such group.
419    /// A snapshot tagged with a group that isn't complete is not one to
420    /// use: its group failed, or is still being taken.
421    pub async fn group_complete(&self, gid: u128) -> Result<Option<bool>, ClientError> {
422        Ok(self
423            .lookup(META_POOL, NS_META, &group_key(gid), b"", None, 0)
424            .await?
425            .map(|l| attr(&l, STATE) == Some(b"complete".as_slice())))
426    }

428    /// Deletes group `gid`: every member's snapshot tagged with it, then the
429    /// group. Also how a group that stopped half-way is cleaned up.
430    pub async fn delete_group(self: &Rc<Self>, gid: u128) -> Result<(), ClientError> {
431        for id in self.group_members(gid).await? {
432            let Some(v) = self.volume(id).await? else {
433                continue;
434            };
435            for s in v.snapshots.iter().filter(|s| s.group == gid) {
436                self.delete_snapshot(id, s.epoch).await?;
437            }
438        }
439        self.exec(META_POOL, NS_META, &group_key(gid), &[WireOp::Delete])
440            .await
441    }

443    /// The volumes of consistency group `gid`.
444    pub(crate) async fn group_members(&self, gid: u128) -> Result<Vec<u128>, ClientError> {
445        let Some(l) = self
446            .lookup(META_POOL, NS_META, &group_key(gid), b"", None, 0)
447            .await?
448        else {
449            return Err(ClientError::Refused(format!("no group {gid:032x}")));
450        };
451        Ok(attr(&l, MEMBERS)
452            .unwrap_or_default()
453            .as_chunks::<16>()
454            .0
455            .iter()
456            .map(|c| u128::from_be_bytes(*c))
457            .collect())
458    }

460    /// Deletes group `gid`'s object once none of its members has a
461    /// snapshot of it left (a purge of its last one).
462    pub(crate) async fn forget_group_if_empty(&self, gid: u128) -> Result<(), ClientError> {
463        let Ok(members) = self.group_members(gid).await else {

479    /// Whether a clone, a rollback or another snapshot of `v` reads
480    /// through snapshot `s`.
481    pub(crate) async fn snapshot_used(
482        &self,
483        v: &Volume,
484        s: &Snapshot,
485    ) -> Result<bool, ClientError> {
486        let branches = self.branches(v.lineage).await?;
487        let reading = self.reading(&branches).await?;
488        let at = (s.branch, s.epoch);
489        Ok(reading.iter().any(|r| r.chain.contains(&at))
490            || v.chain.contains(&at)
491            || v.snapshots
492                .iter()
493                .any(|x| x.epoch != s.epoch && x.chain.contains(&at)))
494    }

496    /// Runs `f` for every chunk a volume of `v`'s size can have, 32 at a
497    /// time.
498    async fn each_chunk<F, Fut>(self: &Rc<Self>, v: &Volume, f: F) -> Result<(), ClientError>
499    where
500        F: Fn(Rc<Client>, u64) -> Fut,
501        Fut: std::future::Future<Output = Result<(), ClientError>> + 'static,
502    {
503        let chunk = u64::from(
504            self.map()
505                .pools
506                .get(&v.pool)
507                .ok_or(ClientError::NoPool(v.pool))?
508                .chunk_size,
509        );
510        let chunks = v.size.div_ceil(chunk);
511        let mut next = 0;
512        while next < chunks {
513            let batch: Vec<_> = (next..(next + 32).min(chunks))
514                .map(|c| hyperstor_reactor::spawn(f(self.clone(), c)))
515                .collect();
516            for b in batch {
517                b.await?;
518            }
519            next += 32;
520        }
521        Ok(())
522    }

563    /// Deletes the snapshot at `epoch` (10-volumes.md §10.3). Refused while a
564    /// clone or rollback depends on it. Its versions are merged into the
565    /// next newer version of each chunk without copying (D104), unless a
566    /// later cut point (a snapshot, a clone's or rollback's parent, the
567    /// head) still reads them through; versions nothing reads are deleted.
568    /// A delete that stops half-way is resumed by deleting it again.
569    pub async fn delete_snapshot(self: &Rc<Self>, id: u128, epoch: u64) -> Result<(), ClientError> {
570        let v = self.finish_trims(id).await?;
571        let s = v
572            .snapshot(epoch)
573            .cloned()
574            .ok_or(ClientError::NoSnapshot(epoch))?;
575        if v.snapshots.iter().any(|x| x.deleting && x.epoch != epoch) {
576            return Err(ClientError::Refused(
577                "another snapshot of this volume is being deleted".into(),
578            ));
579        }
580        // Marked first, then dependents looked for: a clone records itself
581        // before it checks the snapshot, so one of the two always sees the
582        // other.
583        let v = self
584            .change(
585                id,
586                |v| {
587                    let mut n = v.clone();
588                    let s = n
589                        .snapshots
590                        .iter_mut()
591                        .find(|x| x.epoch == epoch)
592                        .ok_or(ClientError::NoSnapshot(epoch))?;
593                    s.deleting = true;
594                    Ok(n)
595                },
596                &[],
597            )
598            .await?;
599        let branches = self.branches(v.lineage).await?;
600        let reading = self.reading(&branches).await?;
601        let used = reading.iter().any(|r| r.chain.contains(&(s.branch, epoch)))
602            || v.chain.contains(&(s.branch, epoch))
603            || v.snapshots
604                .iter()
605                .any(|x| x.epoch != epoch && x.chain.contains(&(s.branch, epoch)));
606        if used {
607            self.change(
608                id,
609                |v| {
610                    let mut n = v.clone();
611                    if let Some(s) = n.snapshots.iter_mut().find(|x| x.epoch == epoch) {
612                        s.deleting = false;
613                    }
614                    Ok(n)
615                },
616                &[],
617            )
618            .await?;
619            return Err(ClientError::SnapshotInUse(epoch));
620        }
621        // What else reads this branch at or below some epoch.
622        let cuts = Self::cuts(&v, &reading, s.branch, epoch);
623        let below = cuts.iter().copied().filter(|c| *c < epoch).max();
624        let above = cuts.iter().copied().filter(|c| *c > epoch).min();
625        let v2 = v.clone();
626        self.each_chunk(&v, move |me, chunk| {
627            let v = v2.clone();
628            async move {
629                let versions = me.versions(&v, chunk, s.branch).await?;
630                let mine: Vec<u64> = versions
631                    .iter()
632                    .copied()
633                    .filter(|e| *e <= epoch && below.is_none_or(|b| *e > b))
634                    .collect();
635                if mine.is_empty() {
636                    return Ok(());
637                }
638                let newer = versions.iter().copied().find(|e| *e > epoch);
639                match (newer, above) {
640                    // The next version absorbs them, newest first, so the
641                    // newest of them wins where they overlap.
642                    (Some(n), a) if a.is_none_or(|a| a >= n) => {
643                        let ops: Vec<WireOp> =
644                            mine.iter().rev().map(|e| WireOp::Absorb(*e)).collect();
645                        let route = Route::Chunk {
646                            lineage: v.lineage,
647                            chunk,
648                            branch: s.branch,
649                            epoch: n,
650                        };
651                        me.exec_at(v.pool, route, &ops).await
652                    }
653                    // Nothing newer, and nothing reads them: gone.
654                    (None, None) => {
655                        let from = below.map_or(0, |b| b + 1);
656                        me.drop_versions(v.pool, v.lineage, chunk, s.branch, from, epoch)
657                            .await
658                    }
659                    // A later cut point still reads them.
660                    _ => Ok(()),
661                }
662            }
663        })
664        .await?;
665        self.change(
666            id,
667            |v| {
668                let mut n = v.clone();
669                n.snapshots.retain(|x| x.epoch != epoch);
670                Ok(n)
671            },
672            &[],
673        )
674        .await?;
675        Ok(())
676    }

678    /// Creates volume `new_id` (or a random id) as a clone of the snapshot
679    /// at `epoch` of volume `id` (10-volumes.md §10.4): a new branch of the
680    /// same lineage (D21), allocating nothing.
681    pub async fn clone_snapshot(
682        &self,
683        id: u128,
684        epoch: u64,
685        name: &str,
686        new_id: Option<u128>,
687    ) -> Result<Volume, ClientError> {
688        let v = self.volume(id).await?.ok_or(ClientError::NoVolume(id))?;
689        // Not one in the trash (D54).
690        let s = v
691            .snapshot(epoch)
692            .filter(|s| !s.deleting && s.trashed_unix == 0)
693            .cloned()
694            .ok_or(ClientError::NoSnapshot(epoch))?;
695        let chain: Vec<(u32, u64)> = std::iter::once((s.branch, epoch))
696            .chain(s.chain.iter().copied())
697            .collect();
698        if chain.len() > MAX_CHAIN {
699            return Err(ClientError::Refused(format!(
700                "clone chains are limited to {MAX_CHAIN} levels: flatten first"
701            )));
702        }
703        let nid =
704            new_id.unwrap_or_else(|| (u128::from(random_u64()) << 64) | u128::from(random_u64()));
705        let b = self.new_branch(v.lineage).await?;
706        let rec = Branch {
707            branch: b,
708            volume: nid,
709            chain: chain.clone(),
710            flattened: 0,
711        };
712        self.record_branch(v.lineage, &rec).await?;
713        // Recorded first, then the snapshot checked again: a delete that
714        // marked it meanwhile wins, and this clone gives up.
715        let still = self.volume(id).await?.is_some_and(|v| {
716            v.snapshot(epoch)
717                .is_some_and(|s| !s.deleting && s.trashed_unix == 0)
718        });
719        if !still {
720            self.forget_branch(v.lineage, b).await?;
721            return Err(ClientError::NoSnapshot(epoch));
722        }
723        let c = Volume {
724            id: nid,
725            pool: v.pool,
726            size: v.size,
727            lineage: v.lineage,
728            state: VolumeState::Active,
729            name: name.to_string(),
730            version: 1,
731            holder: None,
732            delegates: Vec::new(),
733            holder_label: None,
734            head_epoch: 1,
735            branch: b,
736            chain,
737            snapshots: Vec::new(),
738            skip: Vec::new(),
739            incarnation: v.incarnation,
740            trashed_unix: 0,
741        };
742        if let Err(e) = self.put_new(&c).await {
743            self.forget_branch(v.lineage, b).await?;
744            return Err(e);
745        }
746        Ok(c)
747    }

749    /// Rolls volume `id` back to its snapshot at `epoch` (10-volumes.md
750    /// §10.3): its head becomes a new branch cut from the snapshot, at
751    /// once; then the old head's versions newer than anything still reading
752    /// them are deleted. The volume must be detached.
753    pub async fn rollback(self: &Rc<Self>, id: u128, epoch: u64) -> Result<Volume, ClientError> {
754        let v = self.finish_trims(id).await?;
755        if v.holder.is_some() {
756            return Err(ClientError::VolumeAttached(id));
757        }
758        let s = v
759            .snapshot(epoch)
760            .filter(|s| !s.deleting && s.trashed_unix == 0)
761            .cloned()
762            .ok_or(ClientError::NoSnapshot(epoch))?;
763        // To the newest snapshot of the head's branch, which reads through
764        // the same chain as the head: in place (D106). (A flatten since the
765        // snapshot cleared the head's chain, not the snapshot's: then a new
766        // branch, cut from the snapshot with its chain.)
767        let newest = |v: &Volume| {
768            v.snapshots
769                .iter()
770                .filter(|x| x.branch == v.branch)
771                .map(|x| x.epoch)
772                .max()
773        };
774        if s.branch == v.branch && newest(&v) == Some(epoch) && s.chain == v.chain {
775            let lease_free = [WireOp::AssertAttr {
776                name: crate::volume::LEASE_ATTR.to_vec(),
777                value: None,
778            }];
779            self.change(
780                id,
781                |cur| {
782                    if cur.holder.is_some() {
783                        return Err(ClientError::VolumeAttached(id));
784                    }
785                    if cur.branch != s.branch || newest(cur) != Some(epoch) || cur.chain != s.chain
786                    {
787                        return Err(ClientError::Refused(
788                            "the volume changed; roll back again".into(),
789                        ));
790                    }
791                    let mut n = cur.clone();
792                    n.skip.push((epoch + 1, cur.head_epoch));
793                    n.head_epoch = cur.head_epoch + 1;
794                    Ok(n)
795                },
796                &lease_free,
797            )
798            .await?;
799            return self.finish_trims(id).await;
800        }
801        let chain: Vec<(u32, u64)> = std::iter::once((s.branch, epoch))
802            .chain(s.chain.iter().copied())
803            .collect();
804        if chain.len() > MAX_CHAIN {
805            return Err(ClientError::Refused(format!(
806                "rollbacks and clones nest at most {MAX_CHAIN} levels: flatten first"
807            )));
808        }
809        let b = self.new_branch(v.lineage).await?;
810        self.record_branch(
811            v.lineage,
812            &Branch {
813                branch: b,
814                volume: id,
815                chain: chain.clone(),
816                flattened: 0,
817            },
818        )
819        .await?;
820        let lease_free = [WireOp::AssertAttr {
821            name: crate::volume::LEASE_ATTR.to_vec(),
822            value: None,
823        }];
824        let r = self
825            .change(
826                id,
827                |cur| {
828                    if cur.holder.is_some() {
829                        return Err(ClientError::VolumeAttached(id));
830                    }
831                    if cur.snapshot(epoch).is_none_or(|s| s.deleting) {
832                        return Err(ClientError::NoSnapshot(epoch));
833                    }
834                    let mut n = cur.clone();
835                    n.branch = b;
836                    n.chain = chain.clone();
837                    n.head_epoch = cur.head_epoch + 1;
838                    Ok(n)
839                },
840                &lease_free,
841            )
842            .await;
843        let new = match r {
844            Ok(n) => n,
845            Err(e) => {
846                self.forget_branch(v.lineage, b).await?;
847                return Err(e);
848            }
849        };
850        self.trim_branch(&new, v.branch).await?;
851        Ok(new)
852    }

854    /// Deletes the versions of `old` (one of the volume's former head
855    /// branches) above its newest remaining cut point: they are read by
856    /// nothing.
857    async fn trim_branch(self: &Rc<Self>, v: &Volume, old: u32) -> Result<(), ClientError> {
858        let branches = self.branches(v.lineage).await?;
859        let reading = self.reading(&branches).await?;
860        let keep = Self::cuts(v, &reading, old, u64::MAX).into_iter().max();
861        let from = keep.map_or(0, |k| k + 1);
862        let (pool, lineage) = (v.pool, v.lineage);
863        self.each_chunk(v, move |me, chunk| async move {
864            me.drop_versions(pool, lineage, chunk, old, from, u64::MAX)
865                .await
866        })
867        .await
868    }

870    /// Copies into clone `id`'s own branch every block it still reads from
871    /// its parents, then clears its chain (10-volumes.md §10.4): it then
872    /// depends on nothing, and its parents' snapshots can be deleted.
873    /// Refused while it has snapshots taken before a rollback, which read
874    /// through the chain.
875    pub async fn flatten(self: &Rc<Self>, id: u128) -> Result<Volume, ClientError> {
876        let v = self.finish_trims(id).await?;
877        if v.chain.is_empty() {
878            return Ok(v);
879        }
880        let v2 = v.clone();
881        self.each_chunk(&v, move |me, chunk| {
882            let v = v2.clone();
883            async move {
884                let layers: Vec<(u32, u64, u64)> =
885                    v.chain.iter().map(|(b, e)| (*b, 0, *e)).collect();
886                match me
887                    .to_chunk(
888                        v.pool,
889                        v.lineage,
890                        chunk,
891                        v.branch,
892                        v.head_epoch,
893                        &|to, key| Msg::Flatten {
894                            to,
895                            key,
896                            layers: layers.clone(),
897                        },
898                    )
899                    .await?
900                {
901                    (Msg::ExecOk, _) => Ok(()),
902                    (m, _) => Err(unexpected(m)),
903                }
904            }
905        })
906        .await?;
907        let new = self
908            .change(
909                id,
910                |cur| {
911                    let mut n = cur.clone();
912                    n.chain.clear();
913                    Ok(n)
914                },
915                &[],
916            )
917            .await?;
918        // Its own record: flattened at the head epoch the copies went to.
919        let mut branches = self.branches(v.lineage).await?;
920        if let Some(r) = branches.iter_mut().find(|r| r.branch == v.branch) {
921            r.flattened = v.head_epoch;
922            let r = r.clone();
923            self.record_branch(v.lineage, &r).await?;
924        }
925        // Former branches of its own (left by rollbacks) that nothing reads
926        // any more are dropped, until none is left to drop.
927        loop {
928            let reading = self.reading(&branches).await?;
929            let new = self.volume(id).await?.ok_or(ClientError::NoVolume(id))?;
930            let gone = v.chain.iter().map(|(b, _)| *b).find(|b| {
931                *b != new.branch
932                    && branches.iter().any(|r| r.branch == *b && r.volume == id)
933                    && !new
934                        .snapshots
935                        .iter()
936                        .any(|s| s.branch == *b || s.chain.iter().any(|(x, _)| x == b))
937                    && !reading
938                        .iter()
939                        .any(|r| r.branch != *b && r.chain.iter().any(|(x, _)| x == b))
940            });
941            let Some(b) = gone else { break };
942            let (pool, lineage) = (v.pool, v.lineage);
943            self.each_chunk(&v, move |me, chunk| async move {
944                me.drop_versions(pool, lineage, chunk, b, 0, u64::MAX).await
945            })
946            .await?;
947            self.forget_branch(v.lineage, b).await?;
948            branches.retain(|r| r.branch != b);
949        }
950        Ok(new)
951    }

953    /// Deletes the versions a reset in place left (D106), then clears its
954    /// record of them. Run before anything that takes a snapshot, reads a
955    /// branch's versions as a whole, or resets again; a reset that stopped
956    /// half-way is finished here. Returns the volume as it then is.
957    pub(crate) async fn finish_trims(self: &Rc<Self>, id: u128) -> Result<Volume, ClientError> {
958        let v = self.volume(id).await?.ok_or(ClientError::NoVolume(id))?;
959        for (lo, hi) in v.skip.clone() {
960            let (pool, lineage, b) = (v.pool, v.lineage, v.branch);
961            self.each_chunk(&v, move |me, chunk| async move {
962                me.drop_versions(pool, lineage, chunk, b, lo, hi).await
963            })
964            .await?;
965            self.change(
966                id,
967                |cur| {
968                    let mut n = cur.clone();
969                    n.skip.retain(|r| *r != (lo, hi));
970                    Ok(n)
971                },
972                &[],
973            )
974            .await?;
975        }
976        self.volume(id).await?.ok_or(ClientError::NoVolume(id))
977    }

979    /// The branches volume `v` owns: its head's, branch 0 if it began the
980    /// lineage, and those its rollbacks left.
981    pub(crate) async fn own_branches(
982        &self,
983        v: &Volume,
984    ) -> Result<(Vec<u32>, Vec<Branch>), ClientError> {
985        let branches = self.branches(v.lineage).await?;
986        let mut own: Vec<u32> = branches
987            .iter()
988            .filter(|b| b.volume == v.id)
989            .map(|b| b.branch)
990            .collect();
991        own.push(v.branch);
992        if v.lineage == v.id {
993            own.push(0);
994        }
995        own.sort_unstable();
996        own.dedup();
997        Ok((own, branches))
998    }
