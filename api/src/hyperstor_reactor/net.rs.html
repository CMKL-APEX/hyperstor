9/// How long [`bind_retrying`] waits for an address in use to be let go.
10pub const BIND_RETRY: std::time::Duration = std::time::Duration::from_secs(5);
11
12/// Binds `addr`, waiting up to [`BIND_RETRY`] while it is in use. A
13/// reactor's listener stays open a moment after its process has exited:
14/// the kernel tears an io_uring instance down in the background, and a
15/// pending accept holds the socket until then. A service restarted at
16/// once (a test, `systemctl restart`) would otherwise find its own old
17/// address taken.
18pub fn bind_retrying(addr: SocketAddr) -> io::Result<StdListener> {
19    let until = std::time::Instant::now() + BIND_RETRY;
20    loop {
21        match StdListener::bind(addr) {
22            Err(e) if e.kind() == io::ErrorKind::AddrInUse && std::time::Instant::now() < until => {
23                std::thread::sleep(std::time::Duration::from_millis(50));
24            }
25            r => return r,
26        }
27    }
28}

30/// A listening socket.
31pub struct TcpListener {
32    inner: StdListener,
33}
34
35impl TcpListener {
36    /// Binds `addr` ([`bind_retrying`]: waits a moment if it is in use).
37    pub fn bind(addr: SocketAddr) -> io::Result<Self> {
38        let inner = bind_retrying(addr)?;
39        Ok(Self { inner })
40    }
41
42    pub fn local_addr(&self) -> io::Result<SocketAddr> {
43        self.inner.local_addr()
44    }
45
46    /// Wraps a listener bound elsewhere (for example before the reactor
47    /// thread started, so its address could go into the map first).
48    pub fn from_std(inner: StdListener) -> Self {
49        Self { inner }
50    }
51
52    /// Accepts a connection.
53    pub async fn accept(&self) -> io::Result<(TcpStream, SocketAddr)> {
54        let sqe = opcode::Accept::new(
55            types::Fd(self.inner.as_raw_fd()),
56            std::ptr::null_mut(),
57            std::ptr::null_mut(),
58        )
59        .flags(libc::SOCK_CLOEXEC)
60        .build();
61        let (r, _) = current().submit(sqe, Resource::None).await;
62        let fd = check(r)?;
63        // SAFETY: accept returned a new descriptor that we now own.
64        let s = unsafe { StdStream::from_raw_fd(fd) };
65        let peer = s.peer_addr()?;
66        s.set_nodelay(true)?;
67        Ok((TcpStream { inner: s }, peer))
68    }
69}

77fn sockaddr(addr: SocketAddr) -> Box<(libc::sockaddr_storage, libc::socklen_t)> {
78    // SAFETY: sockaddr_storage is plain data; all-zero is a valid value.
79    let mut st: libc::sockaddr_storage = unsafe { std::mem::zeroed() };
80    let len = match addr {
81        SocketAddr::V4(a) => {
82            let sin = libc::sockaddr_in {
83                sin_family: libc::AF_INET as libc::sa_family_t,
84                sin_port: a.port().to_be(),
85                sin_addr: libc::in_addr {
86                    s_addr: u32::from_ne_bytes(a.ip().octets()),
87                },
88                sin_zero: [0; 8],
89            };
90            // SAFETY: sockaddr_in fits in sockaddr_storage.
91            unsafe { std::ptr::write(&mut st as *mut _ as *mut libc::sockaddr_in, sin) };
92            std::mem::size_of::<libc::sockaddr_in>()
93        }
94        SocketAddr::V6(a) => {
95            let sin6 = libc::sockaddr_in6 {
96                sin6_family: libc::AF_INET6 as libc::sa_family_t,
97                sin6_port: a.port().to_be(),
98                sin6_flowinfo: a.flowinfo(),
99                sin6_addr: libc::in6_addr {
100                    s6_addr: a.ip().octets(),
101                },
102                sin6_scope_id: a.scope_id(),
103            };
104            // SAFETY: sockaddr_in6 fits in sockaddr_storage.
105            unsafe { std::ptr::write(&mut st as *mut _ as *mut libc::sockaddr_in6, sin6) };
106            std::mem::size_of::<libc::sockaddr_in6>()
107        }
108    };
109    Box::new((st, len as libc::socklen_t))
110}

112impl TcpStream {
113    /// Connects to `addr`.
114    pub async fn connect(addr: SocketAddr) -> io::Result<Self> {
115        let domain = if addr.is_ipv4() {
116            libc::AF_INET
117        } else {
118            libc::AF_INET6
119        };
120        // SAFETY: socket returns a new descriptor or -1.
121        let fd = unsafe { libc::socket(domain, libc::SOCK_STREAM | libc::SOCK_CLOEXEC, 0) };
122        if fd < 0 {
123            return Err(io::Error::last_os_error());
124        }
125        // SAFETY: fd is new and owned here; the stream closes it on error.
126        let s = unsafe { StdStream::from_raw_fd(fd) };
127        let sa = sockaddr(addr);
128        let sqe = opcode::Connect::new(
129            types::Fd(s.as_raw_fd()),
130            &sa.0 as *const _ as *const libc::sockaddr,
131            sa.1,
132        )
133        .build();
134        let (r, _) = current().submit(sqe, Resource::Addr(sa)).await;
135        check(r)?;
136        s.set_nodelay(true)?;
137        Ok(Self { inner: s })
138    }
139
140    /// Reads into `buf[..buf.len()]`; returns how many bytes arrived (0 at
141    /// end of stream).
142    pub async fn read(&self, mut buf: Vec<u8>) -> (io::Result<usize>, Vec<u8>) {
143        let sqe = opcode::Recv::new(
144            types::Fd(self.inner.as_raw_fd()),
145            buf.as_mut_ptr(),
146            buf.len() as u32,
147        )
148        .build();
149        let (r, res) = current().submit(sqe, Resource::Buf(buf)).await;
150        let Resource::Buf(buf) = res else {
151            unreachable!()
152        };
153        (check(r).map(|n| n as usize), buf)
154    }
155
156    /// Fills `buf` completely, or fails (with `UnexpectedEof` if the peer
157    /// closes first).
158    pub async fn read_exact(&self, mut buf: Vec<u8>) -> (io::Result<()>, Vec<u8>) {
159        let want = buf.len();
160        let mut got = 0;
161        while got < want {
162            let rest = buf.split_off(got);
163            let (r, rest) = self.read(rest).await;
164            buf.extend_from_slice(&rest[..r.as_ref().map_or(0, |n| *n)]);
165            match r {
166                Ok(0) => return (Err(io::ErrorKind::UnexpectedEof.into()), buf),
167                Ok(n) => got += n,
168                Err(e) => return (Err(e), buf),
169            }
170            buf.resize(want.max(buf.len()), 0);
171            buf.truncate(want);
172        }
173        (Ok(()), buf)
174    }
175
176    /// Writes all of `buf`.
177    pub async fn write_all(&self, buf: Vec<u8>) -> (io::Result<()>, Vec<u8>) {
178        let mut off = 0;
179        let mut buf = buf;
180        while off < buf.len() {
181            let sqe = opcode::Send::new(
182                types::Fd(self.inner.as_raw_fd()),
183                buf[off..].as_ptr(),
184                (buf.len() - off) as u32,
185            )
186            .flags(libc::MSG_NOSIGNAL)
187            .build();
188            let (r, res) = current().submit(sqe, Resource::Buf(buf)).await;
189            let Resource::Buf(b) = res else {
190                unreachable!()
191            };
192            buf = b;
193            match check(r) {
194                Ok(0) => return (Err(io::ErrorKind::WriteZero.into()), buf),
195                Ok(n) => off += n as usize,
196                Err(e) => return (Err(e), buf),
197            }
198        }
199        (Ok(()), buf)
200    }
201
202    pub fn peer_addr(&self) -> io::Result<SocketAddr> {
203        self.inner.peer_addr()
204    }
205
206    /// Shuts down both directions; pending reads on other tasks end.
207    pub fn shutdown(&self) -> io::Result<()> {
208        self.inner.shutdown(std::net::Shutdown::Both)
209    }
210
211    /// The descriptor, for handing a connection to another reactor.
212    pub fn into_std(self) -> StdStream {
213        self.inner
214    }
215
216    pub fn from_std(s: StdStream) -> Self {
217        Self { inner: s }
218    }
219
220    /// Duplicates the socket so one task can read while another writes.
221    pub fn try_clone(&self) -> io::Result<Self> {
222        Ok(Self {
223            inner: self.inner.try_clone()?,
224        })
225    }

239    #[test]
240    fn echo_many_clients() {
241        block_on(async {
242            let l = TcpListener::bind("127.0.0.1:0".parse().unwrap()).unwrap();
243            let addr = l.local_addr().unwrap();
244            spawn(async move {
245                loop {
246                    let (s, _) = l.accept().await.unwrap();
247                    spawn(async move {
248                        loop {
249                            let (r, buf) = s.read(vec![0; 4096]).await;
250                            let n = r.unwrap();
251                            if n == 0 {
252                                break;
253                            }
254                            let (w, _) = s.write_all(buf[..n].to_vec()).await;
255                            w.unwrap();
256                        }
257                    });
258                }
259            });
260            let mut hs = Vec::new();
261            for c in 0..20u8 {
262                hs.push(spawn(async move {
263                    let s = TcpStream::connect(addr).await.unwrap();
264                    for i in 0..50u8 {
265                        let msg = vec![c ^ i; 1000];
266                        let (w, _) = s.write_all(msg.clone()).await;
267                        w.unwrap();
268                        let (r, back) = s.read_exact(vec![0; 1000]).await;
269                        r.unwrap();
270                        assert_eq!(back, msg);
271                    }
272                }));
273            }
274            for h in hs {
275                h.await;
276            }
277        });
278    }

280    #[test]
281    fn connect_refused_and_dropped_reads() {
282        block_on(async {
283            let l = StdListener::bind("127.0.0.1:0").unwrap();
284            let addr = l.local_addr().unwrap();
285            drop(l);
286            assert!(TcpStream::connect(addr).await.is_err());
287
288            // A read dropped while pending must not break the reactor.
289            let l = TcpListener::bind("127.0.0.1:0".parse().unwrap()).unwrap();
290            let addr = l.local_addr().unwrap();
291            let c = TcpStream::connect(addr).await.unwrap();
292            let (s, _) = l.accept().await.unwrap();
293            let r = crate::time::timeout(std::time::Duration::from_millis(10), s.read(vec![0; 64]))
294                .await;
295            assert!(r.is_err(), "nothing was sent");
296            let (w, _) = c.write_all(b"after".to_vec()).await;
297            w.unwrap();
298            let (r, buf) = s.read(vec![0; 64]).await;
299            assert_eq!(&buf[..r.unwrap()], b"after");
300        });
301    }

305mod bind_tests {
306    /// An address let go a moment later is bound once it is free.
307    #[test]
308    fn a_bind_waits_for_an_address_let_go() {
309        let held = std::net::TcpListener::bind("127.0.0.1:0").unwrap();
310        let addr = held.local_addr().unwrap();
311        let t = std::thread::spawn(move || {
312            std::thread::sleep(std::time::Duration::from_millis(200));
313            drop(held);
314        });
315        let start = std::time::Instant::now();
316        super::bind_retrying(addr).unwrap();
317        assert!(start.elapsed() >= std::time::Duration::from_millis(150));
318        t.join().unwrap();
319    }
