1//! Waiting on and reading descriptors the reactor doesn't wrap: eventfds
2//! (vhost-user's kick and call, 11-frontends.md §11.1) and Unix sockets whose
3//! messages another library reads once they are ready.
4
5use crate::{Resource, check, current};
6use io_uring::{opcode, types};
7use std::io;
8use std::os::fd::RawFd;
9
10/// Resolves once `fd` has something to read (or is closed or broken).
11/// `fd` must stay open until then.
12pub async fn readable(fd: RawFd) -> io::Result<()> {
13    let sqe = opcode::PollAdd::new(types::Fd(fd), libc::POLLIN as u32).build();
14    let (r, _) = current().submit(sqe, Resource::None).await;
15    check(r).map(|_| ())
16}
17
18/// Reads into `buf[..buf.len()]` from the current position (an eventfd's
19/// 8-byte counter, say); returns how many bytes arrived.
20pub async fn read(fd: RawFd, mut buf: Vec<u8>) -> (io::Result<usize>, Vec<u8>) {
21    let sqe = opcode::Read::new(types::Fd(fd), buf.as_mut_ptr(), buf.len() as u32)
22        .offset(u64::MAX)
23        .build();
24    let (r, res) = current().submit(sqe, Resource::Buf(buf)).await;
25    let Resource::Buf(buf) = res else {
26        unreachable!()
27    };
28    (check(r).map(|n| n as usize), buf)
29}
30
31/// Writes `buf` at the current position; returns how many bytes went.
32pub async fn write(fd: RawFd, buf: Vec<u8>) -> (io::Result<usize>, Vec<u8>) {
33    let sqe = opcode::Write::new(types::Fd(fd), buf.as_ptr(), buf.len() as u32)
34        .offset(u64::MAX)
35        .build();
36    let (r, res) = current().submit(sqe, Resource::Buf(buf)).await;
37    let Resource::Buf(buf) = res else {
38        unreachable!()
39    };
40    (check(r).map(|n| n as usize), buf)
41}

43#[cfg(test)]
44mod tests {
45    use super::*;
46    use std::os::fd::{AsRawFd, FromRawFd, OwnedFd};
47
48    #[test]
49    fn an_eventfd_wakes_a_waiting_task() {
50        // SAFETY: eventfd returns a new descriptor (checked), owned below.
51        let efd = unsafe { libc::eventfd(0, libc::EFD_CLOEXEC) };
52        assert!(efd >= 0);
53        // SAFETY: `efd` is a fresh descriptor nothing else owns.
54        let efd = unsafe { OwnedFd::from_raw_fd(efd) };
55        let fd = efd.as_raw_fd();
56        crate::block_on(async move {
57            let waiter = crate::spawn(async move {
58                readable(fd).await.unwrap();
59                let (r, b) = read(fd, vec![0; 8]).await;
60                assert_eq!(r.unwrap(), 8);
61                u64::from_ne_bytes(b.try_into().unwrap())
62            });
63            crate::time::sleep(std::time::Duration::from_millis(5)).await;
64            assert!(!waiter.is_finished());
65            let (r, _) = write(fd, 3u64.to_ne_bytes().to_vec()).await;
66            assert_eq!(r.unwrap(), 8);
67            assert_eq!(waiter.await, 3);
68        });
69        drop(efd);
70    }
71}
