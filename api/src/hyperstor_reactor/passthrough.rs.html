1//! io_uring passthrough commands (`IORING_OP_URING_CMD`): drivers that take
2//! their commands through io_uring, as ublk does (`hyperstor-ublk`, D94)
3//! and NVMe passthrough will (D3). Only the 16-byte command of a standard
4//! entry; a driver that needs 80 bytes keeps a ring of its own.
5
6use crate::{Resource, current};
7use io_uring::{opcode, types};
8use std::os::fd::RawFd;
9
10/// Sends command `cmd_op` with the 16 bytes `cmd` to the driver behind
11/// `fd`, from this reactor's thread; resolves to the completion's result.
12///
13/// # Safety
14///
15/// `fd` must stay open until the command completes, and any memory the
16/// command names (a buffer address in `cmd`) must stay valid until then,
17/// even if this future is dropped first: the kernel may still use it.
18pub async unsafe fn uring_cmd16(fd: RawFd, cmd_op: u32, cmd: [u8; 16]) -> i32 {
19    let sqe = opcode::UringCmd16::new(types::Fd(fd), cmd_op)
20        .cmd(cmd)
21        .build();
22    let (r, _) = current().submit(sqe, Resource::None).await;
23    r
24}
